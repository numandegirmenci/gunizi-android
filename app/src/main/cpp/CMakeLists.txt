# GunIzi AI 3 native runtime — CallVault pinned GPL-3.0 source, unmodified JNI.
# Source reference: https://github.com/madkongo/CallVault/tree/1561721d21321cf84a251093075b59b351b2e8c0
cmake_minimum_required(VERSION 3.22.1)
project(gunizi_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(GUNIZI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CALLVAULT_ROOT "${GUNIZI_ROOT}/third_party/callvault")
set(WHISPER_ROOT "${CALLVAULT_ROOT}/third_party/whisper.cpp")
set(LLAMA_ROOT "${CALLVAULT_ROOT}/third_party/llama.cpp")
if(NOT EXISTS "${WHISPER_ROOT}/CMakeLists.txt" OR NOT EXISTS "${LLAMA_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "CallVault recursive git submodules missing. Use git submodule update --init --recursive")
endif()
add_compile_options($<$<CONFIG:Debug>:-O3>)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/whisper.cpp")
find_library(log-lib log)
add_library(whispercv SHARED "${CALLVAULT_ROOT}/app/src/main/cpp/whispercv.cpp")
target_link_libraries(whispercv PRIVATE whisper ${log-lib})
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/llama.cpp")
add_library(llamacv SHARED "${CALLVAULT_ROOT}/app/src/main/cpp/llamacv.cpp")
target_link_libraries(llamacv PRIVATE llama ${log-lib})
